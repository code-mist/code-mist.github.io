class Calculator: “““一个简单的计算器类。

此类提供了基本的数学运算功能。

Attributes:
    memory (float): 计算器的记忆存储值
   
    model (str): 计算器型号名称
"""

def __init__(self, model: str = "Standard"):
    """初始化计算器实例。
    
    Args:
        model (str): 计算器的型号名称，默认为"Standard"
    """
    self.memory = 0.0
    self.model = model

def multiply(self, x: float, y: float) -> float:
    """计算两个数的乘积。
    
    此方法将两个数相乘，并将结果存储在内存中。
    
    Args:
        x (float): 第一个因数
        y (float): 第二个因数
        
    Returns:
        float: 两个参数的乘积
        
    Note:
        此操作会更新memory属性为计算结果
    """
    result = x * y
   
    self.memory = result
    return result

def get_memory(self) -> float:
    """获取当前存储在内存中的值。
    
    Returns:
        float: memory属性的当前值
    """
    return self.memory
